    function ThaiBahtText(number) {

      if (isNaN(number) || number === 0) return "ศูนย์บาทถ้วน";

      let str = Math.floor(number).toString();

      const thText = ["ศูนย์", "หนึ่ง", "สอง", "สาม", "สี่", "ห้า", "หก", "เจ็ด", "แปด", "เก้า"];

      const thPos = ["", "สิบ", "ร้อย", "พัน", "หมื่น", "แสน", "ล้าน"];

      let result = ""; let len = str.length;

      for (let i = 0; i < len; i++) {

          let digit = parseInt(str.charAt(i)); let pos = len - 1 - i;

          if (digit !== 0) {

              if (pos === 1 && digit === 1) { result += thPos[pos]; } 

              else if (pos === 1 && digit === 2) { result += "ยี่" + thPos[pos]; } 

              else if (pos === 0 && digit === 1 && len > 1) { result += "เอ็ด"; } 

              else { result += thText[digit] + thPos[pos]; }

          }

      }

      return `(${result}บาทถ้วน)`;

    }

    function updateDashboardStats() {

       const monthInput = document.getElementById('revenue-month').value; 

       if(!monthInput) return; 

       const [year, month] = monthInput.split('-'); 

       

       let totalRev = 0, totalDep = 0, totalBookings = 0, completedJobs = 0;

       let statusCount = { pending: 0, confirmed: 0, completed: 0, cancelled: 0 };

       let packageCount = {};


       allBookings.forEach(b => { 

          // เช็คว่าเป็นข้อมูลในเดือนและปีที่เลือกหรือไม่

          if(b.date && b.date.startsWith(`${year}-${month}`)) { 

             

             // 1. นับจำนวนคิวที่ไม่ใช่ยกเลิก

             if (b.status !== "ยกเลิก") totalBookings++;

             

             // 2. แยกตามสถานะ

             if(b.status === "รอตรวจสอบ") statusCount.pending++;

             else if(b.status === "ยืนยันแล้ว") statusCount.confirmed++;

             else if(b.status === "เสร็จสิ้น") { statusCount.completed++; completedJobs++; }

             else if(b.status === "ยกเลิก") statusCount.cancelled++;


             // 3. คำนวณรายได้ (เฉพาะงานที่คอนเฟิร์มแล้ว)

             if(b.status === "ยืนยันแล้ว" || b.status === "เสร็จสิ้น") { 

                totalRev += Number(b.price) || 0; 

                totalDep += Number(b.depositAmount) || 0; 

             }

             

             // 4. คำนวณแพ็กเกจยอดฮิต (ตัดราคาด้านหลังออกเพื่อให้จัดกลุ่มได้ดีขึ้น)

             if(b.status !== "ยกเลิก" && b.packageName) {

                let pkgName = b.packageName.split('(')[0].trim();

                if(!pkgName) pkgName = b.packageName;

                packageCount[pkgName] = (packageCount[pkgName] || 0) + 1;

             }

          } 

       }); 

       

       // อัปเดตตัวเลขบนหน้าจอ

       document.getElementById('stat-rev-total').innerText = totalRev.toLocaleString(); 

       document.getElementById('stat-rev-deposit').innerText = totalDep.toLocaleString(); 

       document.getElementById('stat-total-bookings').innerText = totalBookings;

       document.getElementById('stat-completed-jobs').innerText = completedJobs;

       

       document.getElementById('stat-status-pending').innerText = statusCount.pending;

       document.getElementById('stat-status-confirmed').innerText = statusCount.confirmed;

       document.getElementById('stat-status-completed').innerText = statusCount.completed;

       document.getElementById('stat-status-cancelled').innerText = statusCount.cancelled;

       

       // สร้าง List แพ็กเกจยอดฮิต

       const popPkgEl = document.getElementById('stat-popular-packages');

       let sortedPkgs = Object.keys(packageCount).map(k => ({ name: k, count: packageCount[k] })).sort((a,b) => b.count - a.count);

       

       if(sortedPkgs.length === 0) {

          popPkgEl.innerHTML = '<li class="text-xs" style="color: var(--text-muted);">ยังไม่มีข้อมูลการจองในเดือนนี้</li>';

       } else {

          popPkgEl.innerHTML = sortedPkgs.slice(0, 3).map((p, idx) => `

             <li class="flex justify-between items-center text-xs">

                <span class="truncate pr-2" style="color: var(--text-main);"><span class="text-accent font-bold mr-1">#${idx+1}</span> ${p.name}</span>

                <span class="font-bold px-2 py-0.5 rounded text-[10px]" style="background: rgba(0,0,0,0.2); color: var(--text-muted);">${p.count} งาน</span>

             </li>

          `).join('');

       }

    }

    function ThaiBahtText(number) {

      if (isNaN(number) || number === 0) return "ศูนย์บาทถ้วน";

      let str = Math.floor(number).toString();

      const thText = ["ศูนย์", "หนึ่ง", "สอง", "สาม", "สี่", "ห้า", "หก", "เจ็ด", "แปด", "เก้า"];

      const thPos = ["", "สิบ", "ร้อย", "พัน", "หมื่น", "แสน", "ล้าน"];

      let result = ""; let len = str.length;

      for (let i = 0; i < len; i++) {

          let digit = parseInt(str.charAt(i)); let pos = len - 1 - i;

          if (digit !== 0) {

              if (pos === 1 && digit === 1) { result += thPos[pos]; } 

              else if (pos === 1 && digit === 2) { result += "ยี่" + thPos[pos]; } 

              else if (pos === 0 && digit === 1 && len > 1) { result += "เอ็ด"; } 

              else { result += thText[digit] + thPos[pos]; }

          }

      }

      return `(${result}บาทถ้วน)`;

    }

    function updateDashboardStats() {

       const monthInput = document.getElementById('revenue-month').value; 

       if(!monthInput) return; 

       const [year, month] = monthInput.split('-'); 

       

       let totalRev = 0, totalDep = 0, totalBookings = 0, completedJobs = 0;

       let statusCount = { pending: 0, confirmed: 0, completed: 0, cancelled: 0 };

       let packageCount = {};


       allBookings.forEach(b => { 

          // เช็คว่าเป็นข้อมูลในเดือนและปีที่เลือกหรือไม่

          if(b.date && b.date.startsWith(`${year}-${month}`)) { 

             

             // 1. นับจำนวนคิวที่ไม่ใช่ยกเลิก

             if (b.status !== "ยกเลิก") totalBookings++;

             

             // 2. แยกตามสถานะ

             if(b.status === "รอตรวจสอบ") statusCount.pending++;

             else if(b.status === "ยืนยันแล้ว") statusCount.confirmed++;

             else if(b.status === "เสร็จสิ้น") { statusCount.completed++; completedJobs++; }

             else if(b.status === "ยกเลิก") statusCount.cancelled++;


             // 3. คำนวณรายได้ (เฉพาะงานที่คอนเฟิร์มแล้ว)

             if(b.status === "ยืนยันแล้ว" || b.status === "เสร็จสิ้น") { 

                totalRev += Number(b.price) || 0;
